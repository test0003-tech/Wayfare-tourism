   116→const getDiscount = (pkg: Package) => {

   125→
   126→// Deterministic "booked recently" number based on package id
   127→const getBookedRecently = (pkgId: string) => {
   128→  let hash = 0;
   129→  for (let i = 0; i < pkgId.length; i++) {
   130→    hash = ((hash << 5) - hash) + pkgId.charCodeAt(i);
   131→    hash |= 0;
   132→  }
   133→  return 12 + (Math.abs(hash) % 37); // 12-48
   134→};

   570→function PackagesContent() {
   571→  const searchParams = useSearchParams();
   572→  const router = useRouter();
   573→  const [packages, setPackages] = useState<Package[]>([]);
   574→  const [fetchedKey, setFetchedKey] = useState<string>('');
   575→  const [search, setSearch] = useState('');
   576→  const [searchFocused, setSearchFocused] = useState(false);
   577→  const [isVisible, setIsVisible] = useState(false);
   578→  const [showMobileFilters, setShowMobileFilters] = useState(false);
   579→  const gridRef = useRef<HTMLDivElement>(null);
   580→  const { addItem, removeItem, isInWishlist } = useWishlist();
   581→
   582→  // Read filters from URL
   583→  const region = searchParams.get('region') || '';
   584→  const category = searchParams.get('category') || '';
   585→  const duration = searchParams.get('duration') || '';
   586→  const sort = searchParams.get('sort') || 'rating';
   587→
   588→  const filterKey = `${region}-${category}-${duration}-${sort}`;
   589→  const loading = fetchedKey !== filterKey;
   590→
   591→  const updateFilter = useCallback((key: string, value: string) => {
   592→    const params = new URLSearchParams(searchParams.toString());
   593→    if (value) {
   594→      params.set(key, value);
   595→    } else {
   596→      params.delete(key);
   597→    }
   598→    router.push(`/packages?${params.toString()}`, { scroll: false });
   599→  }, [router, searchParams]);
   600→
   601→  const clearFilters = useCallback(() => {
   602→    router.push('/packages', { scroll: false });
   603→  }, [router]);
   604→
   605→  const activeFilterCount = [region, category, duration].filter(Boolean).length;
   606→
   607→  useEffect(() => {
   608→    const params = new URLSearchParams();
   609→    if (region) params.set('region', region);
   610→    if (category) params.set('category', category);
   611→    if (duration) params.set('duration', duration);
   612→
   613→    fetch(`/api/packages?${params.toString()}`)
   614→      .then((res) => res.json())
   615→      .then((data) => {
   616→        let sorted = [...data];
   617→        if (sort === 'price-low') {
   618→          sorted.sort((a, b) => a.price - b.price);
   619→        } else if (sort === 'price-high') {
   620→          sorted.sort((a, b) => b.price - a.price);
   621→        } else {
   622→          sorted.sort((a, b) => b.rating - a.rating);
   623→        }
   624→        setPackages(sorted);
   625→        setFetchedKey(filterKey);
   626→      })
   627→      .catch(() => {
   628→        setFetchedKey(filterKey);
   629→      });
   630→  }, [region, category, duration, sort, filterKey]);
   631→
   632→  useEffect(() => {
   633→    const observer = new IntersectionObserver(
   634→      ([entry]) => { if (entry.isIntersecting) setIsVisible(true); },
   635→      { threshold: 0.05 }

   637→    if (gridRef.current) observer.observe(gridRef.current);
   638→    return () => observer.disconnect();
   639→  }, []);

   641→  // Client-side search filter
   642→  const filteredPackages = packages.filter((pkg) =>
   643→    pkg.name.toLowerCase().includes(search.toLowerCase()) ||

   645→    pkg.destination.country.toLowerCase().includes(search.toLowerCase())
   646→  );

   661→
   662→          {/* ═══ Quick Filter Pills ═══ */}
   663→          <div className="py-4">
   664→            <div className="flex items-center gap-2 mb-3">
   665→              <Flame className="h-4 w-4 text-amber-400" />
   666→              <span className="text-xs font-semibold uppercase tracking-wider text-gray-400">Quick Filters</span>
   667→              <div className="h-px flex-1 bg-gradient-to-r from-white/5 to-transparent" />
   668→            </div>
   669→            <div className="flex flex-wrap gap-2">
   670→              <button
   671→                onClick={() => updateFilter('category', '')}
   672→                className={`rounded-full px-4 py-1.5 text-xs font-medium border transition-all duration-300 ${
   673→                  !category
   674→                    ? 'bg-teal-500/20 text-teal-300 border-teal-500/40 glow-teal'
   675→                    : 'glass text-gray-400 border-white/10 hover:text-gray-200 hover:border-white/20'
   676→                }`}
   677→              >
   678→                ✨ All
   679→              </button>
   680→              {quickFilterCategories.map((cat) => (
   681→                <button
   682→                  key={cat.value}
   683→                  onClick={() => updateFilter('category', category === cat.value ? '' : cat.value)}
   684→                  className={`rounded-full px-4 py-1.5 text-xs font-medium border transition-all duration-300 ${
   685→                    getCategoryPillColor(cat.value, category === cat.value)

   692→          </div>
