   133→function HotelsContent() {
   134→  const searchParams = useSearchParams();
   135→  const { addItem, removeItem, isInWishlist } = useWishlist();
   136→  const [hotels, setHotels] = useState<HotelType[]>([]);
   137→  const [loading, setLoading] = useState(true);
   138→  const [searchQuery, setSearchQuery] = useState('');
   139→  const [category, setCategory] = useState('All');
   140→  const [stars, setStars] = useState('All');
   141→  const [sortBy, setSortBy] = useState('rating-desc');
   142→  const [page, setPage] = useState(1);
   143→  const [isVisible, setIsVisible] = useState(false);
   144→  const [heartAnimation, setHeartAnimation] = useState<string | null>(null);
   145→  const sectionRef = useRef<HTMLDivElement>(null);
   146→
   147→  useEffect(() => {
   148→    const categoryParam = searchParams.get('category');
   149→    if (categoryParam && categories.includes(categoryParam)) {
   150→      queueMicrotask(() => setCategory(categoryParam));
   151→    }
   152→  }, [searchParams]);
   153→
   154→  useEffect(() => {
   155→    queueMicrotask(() => setLoading(true));
   156→    const params = new URLSearchParams();
   157→    if (category && category !== 'All') params.set('category', category.toLowerCase());
   158→    fetch(`/api/hotels?${params.toString()}`)
   159→      .then((res) => res.json())
   160→      .then((data) => {
   161→        setHotels(data);
   162→        setLoading(false);
   163→      })
   164→      .catch((err) => {
   165→        console.error(err);
   166→        setLoading(false);
   167→      });
   168→  }, [category]);
   169→
   170→  useEffect(() => {
   171→    const observer = new IntersectionObserver(
   172→      ([entry]) => { if (entry.isIntersecting) setIsVisible(true); },
   173→      { threshold: 0.02 }
   174→    );
   175→    if (sectionRef.current) observer.observe(sectionRef.current);
   176→    return () => observer.disconnect();
   177→  }, []);
   178→
   179→  const filtered = useMemo(() => {
   180→    let result = [...hotels];
   181→
   182→    if (searchQuery.trim()) {
   183→      const q = searchQuery.toLowerCase();
   184→      result = result.filter(
   185→        (h) =>
   186→          h.name.toLowerCase().includes(q) ||
   187→          h.destination.name.toLowerCase().includes(q) ||
   188→          h.destination.country.toLowerCase().includes(q)
   189→      );
   190→    }
   191→
   192→    if (stars !== 'All') {
   193→      const starNum = parseInt(stars);
   194→      result = result.filter((h) => h.stars >= starNum);
   195→    }
   196→
   197→    switch (sortBy) {
   198→      case 'rating-desc':
   199→        result.sort((a, b) => b.rating - a.rating);
   200→        break;
   201→      case 'rating-asc':
   202→        result.sort((a, b) => a.rating - b.rating);
   203→        break;
   204→      case 'price-asc':
   205→        result.sort((a, b) => a.pricePerNight - b.pricePerNight);
   206→        break;
   207→      case 'price-desc':
   208→        result.sort((a, b) => b.pricePerNight - a.pricePerNight);
   209→        break;
   210→      case 'stars-desc':
   211→        result.sort((a, b) => b.stars - a.stars);
   212→        break;
   213→    }
   214→
   215→    return result;
   216→  }, [hotels, searchQuery, stars, sortBy]);
   217→
   218→  const totalPages = Math.ceil(filtered.length / ITEMS_PER_PAGE);
   219→  const paginated = filtered.slice((page - 1) * ITEMS_PER_PAGE, page * ITEMS_PER_PAGE);
   220→
   221→  useEffect(() => { queueMicrotask(() => setPage(1)); }, [searchQuery, category, stars, sortBy]);
   222→

   229→        name: hotel.name,
   230→        destination: `${hotel.destination.name}, ${hotel.destination.country}`,
   231→        image: hotel.image,
   232→        price: hotel.pricePerNight,
   233→        originalPrice: hotel.originalPrice ?? undefined,
   234→        duration: 'Per Night',
   235→        rating: hotel.rating,
   236→        category: hotel.category,
   237→      };
   238→      addItem(item);
   239→      setHeartAnimation(hotel.id);
   240→      setTimeout(() => setHeartAnimation(null), 600);
   241→    }
   242→  }, [addItem, removeItem, isInWishlist]);
   243→

   245→    <PageTransition>

   250→          title="Hotels & Resorts"
   251→          subtitle="Handpicked luxury hotels, boutique resorts & heritage stays"

   255→        <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">

   262→              const Icon = qf.icon;
   263→              const isActive = category === qf.label;
   264→              return (
   265→                <motion.button

   287→                <span className="text-sm font-semibold text-gray-300">Filter & Search</span>
   288→              </div>
   289→              <button
   290→                onClick={() => {
   291→                  setSearchQuery('');
   292→                  setCategory('All');
   293→                  setStars('All');
   294→                  setSortBy('rating-desc');
   295→                }}

   352→              Showing <span className="text-teal-400 font-semibold">{paginated.length}</span> of{' '}
   353→              <span className="text-teal-400 font-semibold">{filtered.length}</span> hotels
   354→            </p>
   355→            {filtered.length > 0 && (
   356→              <p className="text-xs text-gray-600">
   357→                Page {page} of {totalPages}
   358→              </p>
   359→            )}
   360→          </div>
   361→
   362→          {/* Loading state */}
   363→          {loading ? (
   364→            <div className="grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-3">
   365→              {Array.from({ length: 6 }).map((_, i) => (
   366→                <div key={i} className="glass rounded-2xl overflow-hidden animate-pulse">
   367→                  <div className="aspect-[16/12] bg-white/5" />
   368→                  <div className="p-5 space-y-3">

   383→            <div className="text-center py-20">
   384→              <HotelIcon className="h-12 w-12 text-gray-600 mx-auto mb-4" />
   385→              <h3 className="text-lg font-semibold text-gray-300 mb-2">No hotels found</h3>

   387→              <Button
   388→                variant="outline"
   389→                onClick={() => {
   390→                  setSearchQuery('');
   391→                  setCategory('All');
   392→                  setStars('All');
   393→                  setSortBy('rating-desc');
   394→                }}
   395→                className="border-white/10 text-teal-400 hover:bg-white/5 hover:text-teal-300 rounded-xl"
   396→              >
   397→                Clear Filters
   398→              </Button>
   399→            </div>
   400→          ) : (
   401→            <>
   402→              {/* Hotel Cards Grid */}
   403→              <div ref={sectionRef} className="grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-3">
   404→                {paginated.map((hotel, i) => {
   405→                  const amenities = hotel.amenities.split(',').slice(0, 4);

   425→                            src={hotel.image}
   426→                            alt={hotel.name}
   427→                            className="h-full w-full object-cover transition-transform duration-700 group-hover:scale-110"
   428→                          />
   429→                          <div className="absolute inset-0 bg-gradient-to-t from-gray-950/90 via-gray-950/20 to-transparent" />

   431→                          {/* Best Price Guaranteed badge for featured */}

   433→                            <div className="absolute top-3 left-3 flex items-center gap-1.5 rounded-full bg-gradient-to-r from-amber-500 to-amber-600 px-3 py-1 text-xs font-bold text-gray-950 shadow-lg glow-amber">

   440→                          {discount > 0 && !isFeatured && (

   479→
   480→                        <CardContent className="p-4 sm:p-5">
