   293→function FlightsContent() {
   294→  const searchParams = useSearchParams();
   295→  const [flights, setFlights] = useState<FlightDeal[]>([]);
   296→  const [loading, setLoading] = useState(true);
   297→  const [typeFilter, setTypeFilter] = useState('All');
   298→  const [page, setPage] = useState(1);
   299→  const [isVisible, setIsVisible] = useState(false);
   300→  const [searchQuery, setSearchQuery] = useState('');
   301→  const [featuredOnly, setFeaturedOnly] = useState(false);
   302→  const sectionRef = useRef<HTMLDivElement>(null);
   303→
   304→  useEffect(() => {
   305→    const typeParam = searchParams.get('type');
   306→    if (typeParam && flightTypes.includes(typeParam)) {
   307→      queueMicrotask(() => setTypeFilter(typeParam));
   308→    }
   309→  }, [searchParams]);
   310→
   311→  useEffect(() => {
   312→    queueMicrotask(() => setLoading(true));
   313→    fetch('/api/flights')
   314→      .then((res) => res.json())
   315→      .then((data) => {
   316→        setFlights(data);
   317→        setLoading(false);
   318→      })
   319→      .catch((err) => {
   320→        console.error(err);
   321→        setLoading(false);
   322→      });
   323→  }, []);
   324→
   325→  useEffect(() => {
   326→    const observer = new IntersectionObserver(
   327→      ([entry]) => { if (entry.isIntersecting) setIsVisible(true); },
   328→      { threshold: 0.02 }
   329→    );
   330→    if (sectionRef.current) observer.observe(sectionRef.current);
   331→    return () => observer.disconnect();
   332→  }, []);
   333→
   334→  const filtered = useMemo(() => {
   335→    let result = flights;
   336→    if (typeFilter !== 'All') {
   337→      result = result.filter((f) => {
   338→        const type = f.type?.toLowerCase() || '';
   339→        if (typeFilter === 'Round Trip') return type.includes('round');
   340→        if (typeFilter === 'One Way') return type.includes('one') || type.includes('single');
   341→        return true;
   342→      });
   343→    }
   344→    if (searchQuery.trim()) {
   345→      const q = searchQuery.toLowerCase().trim();
   346→      result = result.filter(
   347→        (f) =>
   348→          f.to.toLowerCase().includes(q) ||
   349→          f.from.toLowerCase().includes(q) ||
   350→          f.airline.toLowerCase().includes(q) ||
   351→          f.description.toLowerCase().includes(q)
   352→      );
   353→    }
   354→    if (featuredOnly) {
   355→      result = result.filter((f) => f.featured);
   356→    }
   357→    return result;
   358→  }, [flights, typeFilter, searchQuery, featuredOnly]);
   359→
   360→  useEffect(() => { queueMicrotask(() => setPage(1)); }, [typeFilter, searchQuery, featuredOnly]);
   361→
   362→  const totalPages = Math.ceil(filtered.length / ITEMS_PER_PAGE);
   363→  const paginated = filtered.slice((page - 1) * ITEMS_PER_PAGE, page * ITEMS_PER_PAGE);
   364→
   365→  const getDiscount = useCallback((flight: FlightDeal) => {
   366→    if (!flight.originalPrice) return 0;
   367→    return Math.round(((flight.originalPrice - flight.price) / flight.originalPrice) * 100);
   368→  }, []);
   369→
   370→  // Determine which flight has the best deal (highest discount)
   371→  const bestDealId = useMemo(() => {
   372→    let bestId = '';
   373→    let bestDiscount = 0;
   374→    flights.forEach((f) => {
   375→      const d = getDiscount(f);
   376→      if (d > bestDiscount) {
   377→        bestDiscount = d;
   378→        bestId = f.id;
   379→      }
   380→    });
   381→    return bestId;
   382→  }, [flights, getDiscount]);
   383→
   384→  // Popular routes: top 3 featured flights by discount
   385→  const popularRoutes = useMemo(() => {
   386→    return flights
   387→      .filter((f) => f.featured)
   388→      .sort((a, b) => getDiscount(b) - getDiscount(a))
   389→      .slice(0, 3);
   390→  }, [flights, getDiscount]);
   391→
   392→  return (
   393→    <PageTransition>
   394→      <div className="bg-gray-950 min-h-screen">
   395→        <PageHero
   396→          badge="Flight Deals"
   397→          badgeIcon={PlaneIcon}
   398→          title="Best Flight Deals"
   399→          subtitle="Affordable round-trip flights from major Indian cities to top destinations worldwide"
   400→          backgroundImage="/images/flights-hero.png"
   401→        />
   402→
   403→        <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   404→          <Breadcrumbs items={[{ label: 'Flights' }]} />
   405→
   406→          {/* Popular Routes Highlight Section */}
   407→          {!loading && popularRoutes.length > 0 && (
   408→            <motion.div
   409→              initial={{ opacity: 0, y: 20 }}
   410→              animate={{ opacity: 1, y: 0 }}
   411→              transition={{ duration: 0.6 }}
   412→              className="mb-8"
   413→            >
   414→              <div className="flex items-center gap-2 mb-4">
   415→                <div className="flex items-center gap-2">
   416→                  <span className="text-xl">✈️</span>
   417→                  <h2 className="text-lg font-bold text-white">Popular Routes</h2>
   418→                </div>
   419→                <div className="flex-1 h-px bg-gradient-to-r from-teal-500/30 to-transparent ml-3" />
   420→                <Badge className="bg-amber-500/10 text-amber-400 border-amber-500/20 text-[10px]">
   421→                  <Flame className="h-2.5 w-2.5 mr-0.5" />
   422→                  Trending
   423→                </Badge>
   424→              </div>
   425→              <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   426→                {popularRoutes.map((flight) => (
   427→                  <PopularRouteCard
   428→                    key={flight.id}
   429→                    from={flight.from}
   430→                    to={flight.to}
   431→                    price={flight.price}
   432→                    airline={flight.airline}
   433→                    discount={getDiscount(flight)}
   434→                  />
   435→                ))}
   436→              </div>
   437→            </motion.div>
   438→          )}
   439→
   440→          {/* Filter Section - Redesigned */}
   441→          <div className="mb-8 glass rounded-2xl p-4 sm:p-6">
   442→            <div className="flex items-center gap-2 mb-4">
   443→              <Filter className="h-4 w-4 text-teal-400" />
   444→              <span className="text-sm font-semibold text-gray-300">Filter Flights</span>
   445→              <div className="flex-1 h-px bg-white/5 ml-2" />
   446→            </div>
   447→
   448→            {/* Search bar */}
   449→            <div className="relative mb-4">
   450→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-500" />
   451→              <Input
   452→                type="text"
   453→                placeholder="Search destinations, airlines..."
   454→                value={searchQuery}
   455→                onChange={(e) => setSearchQuery(e.target.value)}
   456→                className="pl-10 bg-white/5 border-white/10 text-white placeholder:text-gray-500 rounded-xl h-10 focus-visible:border-teal-500/50 focus-visible:ring-teal-500/20"
   457→              />
   458→              {searchQuery && (
   459→                <button
   460→                  onClick={() => setSearchQuery('')}
   461→                  className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-500 hover:text-white text-xs"
   462→                >
   463→                  Clear
   464→                </button>
   465→              )}
   466→            </div>
   467→
   468→            {/* Flight type tabs with plane icons */}
   469→            <div className="flex flex-wrap items-center gap-2 mb-4">
   470→              {flightTypes.map((type) => (
   471→                <button
   472→                  key={type}
   473→                  onClick={() => setTypeFilter(type)}
   474→                  className={`flex items-center gap-2 px-4 py-2.5 rounded-xl text-sm font-medium transition-all duration-300 ${
   475→                    typeFilter === type
   476→                      ? 'bg-gradient-to-r from-teal-500 to-emerald-600 text-white glow-teal shadow-lg shadow-teal-500/20'
   477→                      : 'glass text-gray-400 hover:text-teal-400 hover:bg-white/5'
   478→                  }`}
   479→                >
   480→                  <Plane className={`h-3.5 w-3.5 ${typeFilter === type ? 'text-white' : 'text-gray-500'}`} />
   481→                  {type === 'All' ? 'All Flights' : type}
   482→                </button>
   483→              ))}
   484→            </div>
   485→
   486→            {/* Featured Only toggle */}
   487→            <div className="flex items-center gap-3">
   488→              <div className="flex items-center gap-2">
   489→                <Switch
   490→                  checked={featuredOnly}
   491→                  onCheckedChange={setFeaturedOnly}
   492→                  className="data-[state=checked]:bg-teal-500"
   493→                />
   494→                <span className="text-sm text-gray-400 flex items-center gap-1.5">
   495→                  <Star className="h-3.5 w-3.5 text-amber-400" />
   496→                  Featured Only
   497→                </span>
   498→              </div>
   499→            </div>
   500→          </div>
   501→
   502→          {/* Section header with decoration */}
   503→          <div className="mb-6 flex items-center justify-between">
   504→            <div className="flex items-center gap-2">
   505→              <span className="text-lg">✈️</span>
   506→              <p className="text-sm text-gray-400">
   507→                Showing <span className="text-teal-400 font-semibold">{paginated.length}</span> of{' '}
   508→                <span className="text-teal-400 font-semibold">{filtered.length}</span> flight deals
   509→              </p>
   510→            </div>
   511→            {(searchQuery || featuredOnly || typeFilter !== 'All') && (
   512→              <button
   513→                onClick={() => {
   514→                  setSearchQuery('');
   515→                  setFeaturedOnly(false);
   516→                  setTypeFilter('All');
   517→                }}
   518→                className="text-xs text-teal-400 hover:text-teal-300 transition-colors flex items-center gap-1"
   519→              >
   520→                Clear all filters
   521→                <ChevronRight className="h-3 w-3" />
   522→              </button>
   523→            )}
   524→          </div>
   525→
   526→          {/* Loading state */}
   527→          {loading ? (
   528→            <div className="grid grid-cols-1 gap-4 sm:gap-6 sm:grid-cols-2 lg:grid-cols-3">
   529→              {Array.from({ length: 6 }).map((_, i) => (
   530→                <div key={i} className="glass rounded-2xl overflow-hidden animate-pulse">
   531→                  <div className="h-44 bg-white/5" />
   532→                  <div className="p-5 space-y-3">
   533→                    <div className="flex items-center gap-2">
   534→                      <div className="h-8 w-8 rounded-full bg-white/5" />
   535→                      <div className="h-4 bg-white/5 rounded w-1/3" />
   536→                    </div>
   537→                    <div className="h-3 bg-white/5 rounded w-3/4" />
   538→                    <div className="flex justify-between items-end">
   539→                      <div className="space-y-1">
   540→                        <div className="h-3 bg-white/5 rounded w-1/4" />
   541→                        <div className="h-7 bg-white/5 rounded w-1/2" />
   542→                      </div>
   543→                      <div className="h-9 w-24 bg-white/5 rounded-xl" />
   544→                    </div>
   545→                  </div>
   546→                </div>
   547→              ))}
   548→            </div>
   549→          ) : paginated.length === 0 ? (
   550→            <div className="text-center py-20">
   551→              <div className="w-20 h-20 rounded-full glass mx-auto mb-6 flex items-center justify-center">
   552→                <Plane className="h-8 w-8 text-gray-600" />
   553→              </div>
   554→              <h3 className="text-lg font-semibold text-gray-300 mb-2">No flights found</h3>
   555→              <p className="text-gray-500 mb-6 text-sm">
   556→                {searchQuery
   557→                  ? `No results for "${searchQuery}". Try a different search.`
   558→                  : 'Try adjusting your filters'}
   559→              </p>
   560→              <Button
   561→                variant="outline"
   562→                onClick={() => {
   563→                  setSearchQuery('');
   564→                  setFeaturedOnly(false);
   565→                  setTypeFilter('All');
   566→                }}
   567→                className="border-white/10 text-teal-400 hover:bg-white/5 hover:text-teal-300 rounded-xl"
   568→              >
   569→                Show All Flights
   570→              </Button>
   571→            </div>
   572→          ) : (
   573→            <>
   574→              {/* Flight Cards Grid */}
   575→              <div ref={sectionRef} className="grid grid-cols-1 gap-4 sm:gap-6 sm:grid-cols-2 lg:grid-cols-3">
   576→                {paginated.map((flight, i) => (
   577→                  <FlightCard
   578→                    key={flight.id}
   579→                    flight={flight}
   580→                    index={i}
   581→                    isVisible={isVisible}
   582→                    isBestDeal={flight.id === bestDealId}
   583→                  />
   584→                ))}
   585→              </div>
   586→
   587→              {/* Pagination */}
   588→              <AnimatePresence>
   589→                {totalPages > 1 && (
   590→                  <motion.div
   591→                    initial={{ opacity: 0, y: 20 }}
   592→                    animate={{ opacity: 1, y: 0 }}
   593→                    transition={{ duration: 0.4, delay: 0.2 }}
   594→                    className="mt-10 flex items-center justify-center gap-2"
   595→                  >
   596→                    <Button
   597→                      variant="outline"
   598→                      size="sm"
   599→                      disabled={page === 1}
   600→                      onClick={() => setPage((p) => Math.max(1, p - 1))}
   601→                      className="border-white/10 text-gray-300 hover:bg-white/5 hover:text-teal-400 rounded-xl disabled:opacity-40"
   602→                    >
   603→                      Previous
   604→                    </Button>
   605→
   606→                    <div className="flex items-center gap-1">
   607→                      {Array.from({ length: totalPages }).map((_, idx) => (
   608→                        <Button
   609→                          key={idx}
   610→                          variant={page === idx + 1 ? 'default' : 'outline'}
   611→                          size="sm"
   612→                          onClick={() => setPage(idx + 1)}
   613→                          className={
   614→                            page === idx + 1
   615→                              ? 'bg-gradient-to-r from-teal-500 to-emerald-600 text-white rounded-xl shadow-lg shadow-teal-500/20 min-w-[36px]'
   616→                              : 'border-white/10 text-gray-300 hover:bg-white/5 hover:text-teal-400 rounded-xl min-w-[36px]'
   617→                          }
   618→                        >
   619→                          {idx + 1}
   620→                        </Button>
   621→                      ))}
   622→                    </div>
   623→
   624→                    <Button
   625→                      variant="outline"
   626→                      size="sm"
   627→                      disabled={page === totalPages}
   628→                      onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   629→                      className="border-white/10 text-gray-300 hover:bg-white/5 hover:text-teal-400 rounded-xl disabled:opacity-40"
   630→                    >
   631→                      Next
   632→                    </Button>
   633→                  </motion.div>
   634→                )}
   635→              </AnimatePresence>
   636→            </>
   637→          )}
   638→        </div>
   639→
   640→        {/* Bottom spacer */}
   641→        <div className="h-20" />
   642→      </div>
   643→    </PageTransition>
   644→  );
   645→}
